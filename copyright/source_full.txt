# -*- coding: utf-8 -*-
"""配置加载与校验。"""
from __future__ import annotations

import json
from pathlib import Path

from .decision import Thresholds

DEFAULTS = {
    "cotton_gate": 0.20,
    "cotton_confidence": 0.46,
    "cotton_margin": 0.0,
    "fiber_confidence": 0.50,
    "focus": 75.0,
    "focus_required": False,
    "cotton_input": 384,
    "fiber_input": 288,
    "roi": 660,
}

def load_thresholds(path: str | Path | None = None) -> Thresholds:
    """从 JSON 读取阈值；字段缺失时使用默认值。"""
    data = dict(DEFAULTS)
    if path is not None:
        payload = json.loads(Path(path).read_text(encoding="utf-8"))
        unknown = set(payload) - set(DEFAULTS)
        if unknown:
            raise ValueError(f"未知配置项: {sorted(unknown)}")
        data.update(payload)
    return Thresholds(
        cotton_gate=float(data["cotton_gate"]),
        cotton_confidence=float(data["cotton_confidence"]),
        cotton_margin=float(data["cotton_margin"]),
        fiber_confidence=float(data["fiber_confidence"]),
        focus=float(data["focus"]),
        focus_required=bool(data["focus_required"]),
    )

# -*- coding: utf-8 -*-
"""级联判定：棉概率门控、三档判定与交叉校验。"""
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Sequence

import numpy as np

from .labels import COTTON_TIERS, cotton_business, cotton_index, fiber_business

def softmax(logits: Sequence[float]) -> np.ndarray:
    """数值稳定的 softmax。已经是概率向量时原样归一化。"""
    values = np.asarray(logits, dtype=np.float32).reshape(-1)
    if values.size == 0:
        raise ValueError("logits 不能为空")
    if np.all(values >= -1e-5) and abs(float(values.sum()) - 1.0) < 1e-3:
        return values / max(float(values.sum()), 1e-12)
    values = values - float(values.max())
    exp = np.exp(values)
    return exp / max(float(exp.sum()), 1e-12)

@dataclass
class Thresholds:
    """所有阈值集中管理，便于现场标定。"""

    cotton_gate: float = 0.20
    cotton_confidence: float = 0.46
    cotton_margin: float = 0.0
    fiber_confidence: float = 0.50
    focus: float = 75.0
    focus_required: bool = False

@dataclass
class InspectionResult:
    cotton_tier: str
    cotton_raw: str
    cotton_confidence: float
    cotton_margin: float
    cotton_probabilities: list[float]
    cotton_gate_probability: float
    cotton_entered: bool
    pure_fiber: str
    fiber_raw: str
    fiber_confidence: float
    fiber_probabilities: list[float]
    cross_check: str
    focus: float
    thresholds: dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        return asdict(self)

def decide(cotton_probs, fiber_probs, fiber_names, thresholds: Thresholds,
           focus: float | None = None) -> InspectionResult:
    """执行一次完整判定。

    cotton_probs 为 None 表示未运行棉含量模型；函数会根据 P(棉) 自行决定
    是否需要它。返回结构可直接序列化为 JSON。
    """
    fiber_p = softmax(fiber_probs)
    if len(fiber_p) != len(fiber_names):
        raise ValueError("fiber_probs 长度与 fiber_names 不一致")

    index = cotton_index(fiber_names)
    gate_probability = float(fiber_p[index]) if index is not None else 0.0
    entered = gate_probability >= thresholds.cotton_gate

    fiber_top = int(np.argmax(fiber_p))
    fiber_raw = fiber_names[fiber_top]
    fiber_conf = float(fiber_p[fiber_top])
    pure_fiber = fiber_business(fiber_raw)
    if fiber_conf < thresholds.fiber_confidence:
        pure_fiber = "其他"

    if not entered:
        cotton_tier = "不适用"
        cotton_raw = "未运行"
        cotton_conf = 0.0
        cotton_margin = 0.0
        cotton_p = [0.0, 0.0, 0.0]
        cotton_valid = False
    else:
        if cotton_probs is None:
            raise ValueError("P(棉) 达到门限，但未提供 cotton_probs")
        cotton_p = softmax(cotton_probs)
        if len(cotton_p) != len(COTTON_TIERS):
            raise ValueError("cotton_probs 必须是三分类输出")
        order = np.sort(cotton_p)
        cotton_margin = float(order[-1] - order[-2])
        top = int(np.argmax(cotton_p))
        cotton_raw = COTTON_TIERS[top]
        cotton_conf = float(cotton_p[top])
        cotton_tier = cotton_business(cotton_raw)
        cotton_valid = (cotton_conf >= thresholds.cotton_confidence
                        and cotton_margin >= thresholds.cotton_margin)
        if not cotton_valid:
            cotton_tier = "待定"

    passed = entered and (
        (cotton_tier == "低棉" and pure_fiber == "其他")
        or (cotton_tier == "高棉" and pure_fiber == "纯棉")
    )
    if thresholds.focus_required and focus is not None and focus < thresholds.focus:
        passed = False

    return InspectionResult(
        cotton_tier=cotton_tier,
        cotton_raw=cotton_raw,
        cotton_confidence=round(cotton_conf, 5),
        cotton_margin=round(cotton_margin, 5),
        cotton_probabilities=[round(float(value), 5) for value in cotton_p],
        cotton_gate_probability=round(gate_probability, 5),
        cotton_entered=entered,
        pure_fiber=pure_fiber,
        fiber_raw=fiber_raw,
        fiber_confidence=round(fiber_conf, 5),
        fiber_probabilities=[round(float(value), 5) for value in fiber_p],
        cross_check="通过" if passed else "复核",
        focus=round(float(focus), 2) if focus is not None else 0.0,
        thresholds=asdict(thresholds),
    )

    "Tier1_Mid": "中棉", "混纺": "中棉", "中棉": "中棉",
    "Tier2_High": "高棉", "高棉": "高棉",
}

    "棉": "纯棉", "纯棉": "纯棉",
    "羊毛": "纯羊毛", "纯羊毛": "纯羊毛",
    "涤纶": "纯涤纶", "纯涤纶": "纯涤纶",
    "丝": "其他", "亚麻": "其他", "其他": "其他",
}

def cotton_business(name: str) -> str:
    """把模型原始类别名转换为业务档位。未知类别返回“其他”。"""
    return _TIER_ALIASES.get(name, "其他")

def fiber_business(name: str) -> str:
    """把六类纤维名转换为业务四类。未知类别返回“其他”。"""
    return _FIBER_ALIASES.get(name, "其他")

def cotton_index(fiber_names) -> int | None:
    """返回纯面料模型中“棉”类别的索引；不存在时返回 None。"""
    for index, name in enumerate(fiber_names):
        if name == "棉":
            return index
    return None

# -*- coding: utf-8 -*-
"""评估指标：混淆矩阵、召回率、整布表决。"""
from __future__ import annotations

from collections import Counter, defaultdict

import numpy as np

def confusion_matrix(y_true, y_pred, labels) -> np.ndarray:
    """按 labels 顺序生成混淆矩阵，行=真值，列=预测。"""
    index = {label: i for i, label in enumerate(labels)}
    matrix = np.zeros((len(labels), len(labels)), dtype=int)
    for truth, pred in zip(y_true, y_pred):
        if truth not in index or pred not in index:
            raise ValueError(f"未知类别: {truth}, {pred}")
        matrix[index[truth], index[pred]] += 1
    return matrix

def per_class_recall(matrix: np.ndarray) -> list[float]:
    """逐类召回率；无样本的类别返回 0。"""
    result = []
    for i in range(matrix.shape[0]):
        total = int(matrix[i].sum())
        result.append(float(matrix[i, i] / total) if total else 0.0)
    return result

def top1_accuracy(matrix: np.ndarray) -> float:
    total = int(matrix.sum())
    return float(np.trace(matrix) / total) if total else 0.0

def group_vote_accuracy(groups: dict[str, list[np.ndarray]],
                        group_truth: dict[str, str],
                        labels) -> float:
    """整布表决：同一布料多视角概率取平均后再取最大类。"""
    if not groups:
        return 0.0
    index = {label: i for i, label in enumerate(labels)}
    correct = 0
    for group, probabilities in groups.items():
        mean = np.mean(np.asarray(probabilities, dtype=np.float32), axis=0)
        pred = labels[int(np.argmax(mean))]
        correct += int(pred == group_truth[group])
    return correct / len(groups)

def majority_vote(predictions) -> str:
    """离散标签的多数表决；票数相同时保留首次出现的标签。"""
    if not predictions:
        raise ValueError("predictions 不能为空")
    counts = Counter(predictions)
    best = max(counts.values())
    for label in predictions:
        if counts[label] == best:
            return label
    return predictions[0]

def accumulate(groups: dict[str, list[np.ndarray]], key: str,
               probabilities: np.ndarray) -> None:
    """把一次推理结果追加到指定布料的概率列表。"""
    groups.setdefault(key, []).append(np.asarray(probabilities, dtype=np.float32))

# -*- coding: utf-8 -*-
"""图像预处理：中心 ROI、缩放、归一化和清晰度评估。"""
from __future__ import annotations

import numpy as np

IMAGENET_MEAN = np.array([0.485, 0.456, 0.406], dtype=np.float32)
IMAGENET_STD = np.array([0.229, 0.224, 0.225], dtype=np.float32)

def center_square(image: np.ndarray, side: int | None = None) -> np.ndarray:
    """取图像中心正方形区域。

    side 为 None 时取短边；否则取 min(高, 宽, side)。
    返回的是视图副本，避免后续原地修改影响原图。
    """
    if image.ndim not in (2, 3):
        raise ValueError("image 必须是 HxW 或 HxWxC 数组")
    height, width = image.shape[:2]
    if height <= 0 or width <= 0:
        raise ValueError("图像尺寸必须为正数")
    target = min(height, width) if side is None else min(height, width, side)
    top = (height - target) // 2
    left = (width - target) // 2
    return image[top:top + target, left:left + target].copy()

def resize_square(image: np.ndarray, size: int) -> np.ndarray:
    """把图像缩放到 size×size。

    优先使用 OpenCV 的 INTER_AREA；没有 OpenCV 时退化为最近邻，
    后者只用于无依赖的单元测试，正式推理应安装 opencv。
    """
    if size <= 0:
        raise ValueError("size 必须为正数")
    try:
        import cv2
        return cv2.resize(image, (size, size), interpolation=cv2.INTER_AREA)
    except ImportError:
        height, width = image.shape[:2]
        rows = (np.arange(size) * height / size).astype(np.int32)
        cols = (np.arange(size) * width / size).astype(np.int32)
        return image[rows][:, cols]

def to_float_rgb(image: np.ndarray) -> np.ndarray:
    """BGR uint8 -> RGB float32，范围 0~1。"""
    if image.ndim != 3 or image.shape[2] != 3:
        raise ValueError("image 必须是 HxWx3")
    rgb = image[:, :, ::-1].astype(np.float32) / 255.0
    return rgb

def imagenet_normalize(rgb: np.ndarray) -> np.ndarray:
    """按 ImageNet 均值/方差归一化，返回 HWC float32。"""
    return (rgb - IMAGENET_MEAN) / IMAGENET_STD

def plain_normalize(rgb: np.ndarray) -> np.ndarray:
    """仅做 0~1 归一化，供 YOLO 系模型使用。"""
    return rgb.astype(np.float32)

def focus_score(image: np.ndarray) -> float:
    """用拉普拉斯方差衡量清晰度。

    数值越大表示边缘越锐利。不同相机/倍率下阈值需要重新标定，
    默认参考值为 75。
    """
    gray = image
    if image.ndim == 3:
        # BGR -> 灰度，使用整数权重近似
        gray = (image[:, :, 0] * 0.114 + image[:, :, 1] * 0.587
                + image[:, :, 2] * 0.299)
    gray = gray.astype(np.float32)
    if gray.shape[0] < 3 or gray.shape[1] < 3:
        return 0.0
    kernel = np.array([[0, 1, 0], [1, -4, 1], [0, 1, 0]], dtype=np.float32)
    height, width = gray.shape
    result = np.zeros((height - 2, width - 2), dtype=np.float32)
    for row in range(3):
        for col in range(3):
            result += kernel[row, col] * gray[row:row + height - 2,
                                               col:col + width - 2]
    return float(result.var())

# -*- coding: utf-8 -*-
"""检测台账：CSV 追加、编号续接、图片归档与失败回滚。"""
from __future__ import annotations

import csv
import os
from datetime import datetime
from pathlib import Path

CSV_HEADER = [
    "检测时间", "样品编号", "棉含量档位", "纯面料结果", "交叉结论",
    "棉含量置信度", "纯面料置信度", "棉门控概率", "清晰度", "存样图",
]

class RecordWriter:
    """一个目录只应由一个进程写入。"""

    def __init__(self, folder):
        self.folder = Path(folder)
        self.folder.mkdir(parents=True, exist_ok=True)
        self.report = self.folder / "inspection_report.csv"
        self._sample_id = self._read_last_id() + 1

    def _read_last_id(self) -> int:
        if not self.report.exists() or self.report.stat().st_size == 0:
            return 0
        last = 0
        with self.report.open(newline="", encoding="utf-8-sig") as stream:
            reader = csv.DictReader(stream)
            if reader.fieldnames != CSV_HEADER:
                raise RuntimeError("台账列结构不兼容，请使用新的归档目录")
            for row in reader:
                value = row["样品编号"].rsplit("-", 1)[-1]
                last = max(last, int(value))
        return last

    def allocate(self) -> int:
        sample_id = self._sample_id
        self._sample_id += 1
        return sample_id

    def save(self, result, image_bytes: bytes | None = None,
             sample_id: int | None = None) -> dict:
        """写入一行记录；提供 JPEG 字节时同时保存样图。

        返回写入的字段字典，便于调用方显示或审计。
        """
        sample_id = sample_id or self.allocate()
        timestamp = datetime.now()
        image_name = ""
        image_path = None
        if image_bytes:
            image_name = (f"Sample{sample_id:03d}_"
                          f"{timestamp.strftime('%Y%m%d_%H%M%S_%f')}.jpg")
            image_path = self.folder / image_name
            image_path.write_bytes(image_bytes)

        row = {
            "检测时间": timestamp.isoformat(timespec="seconds"),
            "样品编号": f"Sample-{sample_id:03d}",
            "棉含量档位": result.cotton_tier,
            "纯面料结果": result.pure_fiber,
            "交叉结论": result.cross_check,
            "棉含量置信度": f"{result.cotton_confidence:.2%}",
            "纯面料置信度": f"{result.fiber_confidence:.2%}",
            "棉门控概率": f"{result.cotton_gate_probability:.2%}",
            "清晰度": f"{result.focus:.1f}",
            "存样图": image_name,
        }

        previous_size = self.report.stat().st_size if self.report.exists() else 0
        need_header = previous_size == 0
        try:
            with self.report.open("a", newline="", encoding="utf-8-sig") as stream:
                writer = csv.DictWriter(stream, fieldnames=CSV_HEADER)
                if need_header:
                    writer.writeheader()
                writer.writerow(row)
                stream.flush()
                os.fsync(stream.fileno())
        except Exception:
            if self.report.exists():
                with self.report.open("r+b") as stream:
                    stream.truncate(previous_size)
            if image_path and image_path.exists():
                image_path.unlink()
            raise
        return {"sample_id": sample_id, "image": image_name, "row": row}

# -*- coding: utf-8 -*-
"""NCNN 推理封装与双模型流水线。

该模块与具体操作系统无关，树莓派、x86 或其他 Linux 开发板均可使用，
只要求安装 ncnn、numpy、opencv。
"""
from __future__ import annotations

from pathlib import Path

import numpy as np

from .decision import InspectionResult, Thresholds, decide
from .preprocess import center_square, focus_score, resize_square, to_float_rgb

class NcnnModel:
    """单个 NCNN 分类模型。

    imagenet=True 时使用 ImageNet 均值/方差，否则只做 0~1 归一化。
    """

    def __init__(self, param, binary, names, input_size: int,
                 threads: int = 2, imagenet: bool = True, fp16: bool = True):
        import ncnn

        self.ncnn = ncnn
        self.names = tuple(names)
        self.input_size = int(input_size)
        self.imagenet = bool(imagenet)
        self.net = ncnn.Net()
        self.net.opt.num_threads = int(threads)
        self.net.opt.use_vulkan_compute = False
        self.net.opt.use_fp16_packed = bool(fp16)
        self.net.opt.use_fp16_storage = bool(fp16)
        self.net.opt.use_packing_layout = True
        if self.net.load_param(str(param)) != 0:
            raise RuntimeError(f"NCNN 结构加载失败: {param}")
        if self.net.load_model(str(binary)) != 0:
            raise RuntimeError(f"NCNN 权重加载失败: {binary}")

    def _mat(self, bgr: np.ndarray):
        crop = center_square(bgr, self.input_size)
        crop = resize_square(crop, self.input_size)
        rgb = np.ascontiguousarray(crop[:, :, ::-1])
        mat = self.ncnn.Mat.from_pixels(
            rgb.tobytes(), self.ncnn.Mat.PixelType.PIXEL_RGB,
            self.input_size, self.input_size)
        if self.imagenet:
            mat.substract_mean_normalize(
                [0.485 * 255, 0.456 * 255, 0.406 * 255],
                [1 / (0.229 * 255), 1 / (0.224 * 255), 1 / (0.225 * 255)])
        else:
            mat.substract_mean_normalize([0, 0, 0], [1 / 255, 1 / 255, 1 / 255])
        return mat

    def predict(self, bgr: np.ndarray, tta: bool = False) -> np.ndarray:
        """返回类别概率向量。tta=True 时对 0/90/180/270 度取平均。"""
        frames = [bgr] if not tta else [np.rot90(bgr, k).copy() for k in range(4)]
        probabilities = []
        for frame in frames:
            extractor = self.net.create_extractor()
            if extractor.input("in0", self._mat(frame)) != 0:
                raise RuntimeError("NCNN 输入节点 in0 写入失败")
            code, output = extractor.extract("out0")
            if code != 0:
                raise RuntimeError(f"NCNN 推理失败: {code}")
            values = np.asarray(output, dtype=np.float32).reshape(-1)
            probabilities.append(_softmax(values))
        return np.mean(probabilities, axis=0)

def _softmax(values: np.ndarray) -> np.ndarray:
    values = values - float(values.max())
    exp = np.exp(values)
    return exp / max(float(exp.sum()), 1e-12)

class InspectionPipeline:
    """级联双模型流水线：纯面料门控 -> 棉含量 -> 交叉校验。"""

    def __init__(self, cotton: NcnnModel, fiber: NcnnModel,
                 thresholds: Thresholds | None = None):
        self.cotton = cotton
        self.fiber = fiber
        self.thresholds = thresholds or Thresholds()

    def inspect(self, bgr: np.ndarray, tta: bool = False) -> InspectionResult:
        fiber_probs = self.fiber.predict(bgr, tta=tta)
        focus = focus_score(center_square(bgr, 660))
        from .labels import cotton_index
        index = cotton_index(self.fiber.names)
        gate = float(fiber_probs[index]) if index is not None else 0.0
        cotton_probs = None
        if gate >= self.thresholds.cotton_gate:
            cotton_probs = self.cotton.predict(bgr, tta=tta)
        return decide(cotton_probs, fiber_probs, self.fiber.names,
                      self.thresholds, focus=focus)

    @classmethod
    def from_directory(cls, model_dir, threads: int = 2, fp16: bool = True,
                       thresholds: Thresholds | None = None):
        """从包含 cotton_tier/pure_fiber 的模型目录构建流水线。"""
        import json

        model_dir = Path(model_dir)
        cotton_dir = model_dir / "cotton_tier"
        fiber_dir = model_dir / "pure_fiber"
        cotton_names = json.loads((cotton_dir / "labels.json").read_text(encoding="utf-8"))
        fiber_names = json.loads((fiber_dir / "labels.json").read_text(encoding="utf-8"))
        cotton = NcnnModel(cotton_dir / "model.ncnn.param",
                           cotton_dir / "model.ncnn.bin",
                           cotton_names, 384, threads, True, fp16)
        fiber = NcnnModel(fiber_dir / "model.ncnn.param",
                          fiber_dir / "model.ncnn.bin",
                          fiber_names, 288, threads, False, fp16)
        return cls(cotton, fiber, thresholds)

# -*- coding: utf-8 -*-
"""Build grouped train/val/blind datasets for the two-model fabric pipeline.

The split is made by fabric/sample id, never by individual image.  This is
important because each fabric has several microscope views.
"""
from __future__ import annotations

import argparse
import hashlib
import random
import re
import shutil
from collections import defaultdict
from pathlib import Path

IMAGE_EXTS = {".jpg", ".jpeg", ".png", ".bmp"}
TIERS = ("Tier0_Low", "Tier1_Mid", "Tier2_High")
TIER_BY_RANGE = {"0": "Tier0_Low", "1-50": "Tier0_Low",
                 "50-79": "Tier1_Mid", "80-100": "Tier2_High"}
FIBERS = ("纯棉", "纯羊毛", "纯涤纶", "其他")
RAW_FIBERS = ("丝", "亚麻", "其他", "棉", "涤纶", "羊毛")

def images(folder: Path):
    return [p for p in folder.rglob("*") if p.is_file() and p.suffix.lower() in IMAGE_EXTS]

def group_id(path: Path) -> str:
    # d10_01 and a1_05 are views of one fabric. External files use a stable
    # filename prefix; retaining the parent folder avoids cross-source clashes.
    stem = path.stem
    # Microscope captures use *_01 ... *_05 for one fabric. Public/external
    # images have no view suffix, so each source image remains its own group.
    parts = stem.split("_")
    if len(parts) >= 2 and re.fullmatch(r"\d{2}", parts[-1]):
        prefix = "_".join(parts[:-1])
    else:
        prefix = stem
    return f"{path.parent.name}_{prefix}"

def clear_make(root: Path, classes):
    if root.exists():
        shutil.rmtree(root)
    for split in ("train", "val", "blind"):
        for cls in classes:
            (root / split / cls).mkdir(parents=True, exist_ok=True)

def split_groups(groups, seed, val_ratio=.15, blind_ratio=.20):
    rng = random.Random(seed)
    keys = list(groups)
    rng.shuffle(keys)
    n_blind = max(1, round(len(keys) * blind_ratio))
    n_val = max(1, round((len(keys) - n_blind) * val_ratio))
    return keys[n_blind + n_val:], keys[n_blind:n_blind + n_val], keys[:n_blind]

def copy_one(src: Path, dst: Path):
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src, dst)

def copy_grouped(records, out, classes, seed):
    clear_make(out, classes)
    by_class = {c: defaultdict(list) for c in classes}
    for cls, src in records:
        by_class[cls][group_id(src)].append(src)
    counts = defaultdict(lambda: defaultdict(int))
    for cls in classes:
        train, val, blind = split_groups(by_class[cls], seed)
        for split, keys in (("train", train), ("val", val), ("blind", blind)):
            for key in keys:
                for src in by_class[cls][key]:
                    # Prefixes make collisions impossible while keeping source
                    # and group information visible during error analysis.
                    dst = out / split / cls / f"{key}_{src.name}"
                    shutil.copy2(src, dst)
                    counts[split][cls] += 1
    return counts

def collect_tiers(root: Path):
    records = []
    zijian = root / "mianliao_zijian"
    mapping = TIER_BY_RANGE
    for folder, cls in mapping.items():
        records.extend((cls, p) for p in images(zijian / folder))
    # The public set has explicit cotton percentages in directory names.
    original = root / "CottonFabricImageBD" / "Original"
    for folder in original.iterdir() if original.exists() else []:
        if not folder.is_dir():
            continue
        m = re.search(r"\d+", folder.name)
        if not m:
            continue
        value = int(m.group())
        cls = "Tier0_Low" if value < 50 else "Tier1_Mid" if value < 80 else "Tier2_High"
        records.extend((cls, p) for p in images(folder))
    return records

def copy_cotton_dataset(root: Path, out: Path, seed: int, real_repeat: int = 8):
    """Build a domain-matched cotton split.

    The fixed 16 self-built fabrics are the historical blind set used by the
    project. Public images are training-only; self-built images are repeated
    in training so the model does not optimize for the easier public domain.
    """
    fixed_blind = {"d2", "d7", "d8", "d10", "d12", "d17",
                   "a1", "a6", "a7", "b3", "b5", "b7", "c1", "c4", "c5", "c7"}
    clear_make(out, TIERS)
    src = root / "mianliao_zijian"
    real = {cls: defaultdict(list) for cls in TIERS}
    for folder, cls in TIER_BY_RANGE.items():
        for image in images(src / folder):
            prefix = image.stem.split("_")[0]
            real[cls][f"{folder}_{prefix}"].append(image)

    rng = random.Random(seed)
    counts = defaultdict(lambda: defaultdict(int))
    for cls in TIERS:
        blind_keys = [key for key in real[cls] if key.rsplit("_", 1)[-1] in fixed_blind]
        pool = [key for key in real[cls] if key not in blind_keys]
        rng.shuffle(pool)
        n_val = max(1, round(len(pool) * .20)) if pool else 0
        val_keys, train_keys = pool[:n_val], pool[n_val:]
        for split, keys in (("val", val_keys), ("blind", blind_keys)):
            for key in keys:
                for image in real[cls][key]:
                    copy_one(image, out / split / cls / f"{key}_{image.name}")
                    counts[split][cls] += 1
        for key in train_keys:
            for image in real[cls][key]:
                for repeat in range(real_repeat):
                    suffix = f"__real{repeat}" if repeat else ""
                    copy_one(image, out / "train" / cls / f"{key}_{image.stem}{suffix}{image.suffix}")
                    counts["train"][cls] += 1

    # Every public image is independent training data. Its original folder
    # carries the cotton percentage and is therefore already class-labelled.
    public = root / "CottonFabricImageBD" / "Original"
    for folder in public.iterdir() if public.exists() else []:
        if not folder.is_dir():
            continue
        match = re.search(r"\d+", folder.name)
        if not match:
            continue
        value = int(match.group())
        cls = "Tier0_Low" if value < 50 else "Tier1_Mid" if value < 80 else "Tier2_High"
        for image in images(folder):
            copy_one(image, out / "train" / cls / f"public_{folder.name}_{image.name}")
            counts["train"][cls] += 1
    return counts

def collect_fibers(root: Path):
    records = []
    source = root / "mianliao_5"
    aliases = {"棉": "纯棉", "羊毛": "纯羊毛", "涤纶": "纯涤纶", "丝": "其他", "亚麻": "其他"}
    for folder in source.iterdir() if source.exists() else []:
        cls = aliases.get(folder.name)
        if cls:
            records.extend((cls, p) for p in images(folder))
    # High-cotton, composition-labelled microscope samples are useful pure
    # cotton positives. Keep the mapping explicit so this assumption is easy
    # to replace when laboratory composition labels become available.
    for p in images(root / "mianliao_zijian" / "80-100"):
        records.append(("纯棉", p))
    # Extra non-target material improves rejection, but is never put in the
    # target classes. It is optional and can be omitted on a smaller dataset.
    other = root / "data_other" / "train"
    for p in images(other):
        records.append(("其他", p))
    # The same microscope frame occurs in both m5 and self-built folders under
    # different filenames. Keep only one byte-identical copy so it cannot
    # cross a split boundary or overweight one class.
    seen = set()
    unique = []
    for cls, path in records:
        digest = hashlib.md5(path.read_bytes()).hexdigest()
        if digest in seen:
            continue
        seen.add(digest)
        unique.append((cls, path))
    return unique

def collect_raw_fibers(root: Path):
    """Collect the original six-class labels used by the existing fiber5 model."""
    records = []
    source = root / "mianliao_5"
    for folder in source.iterdir() if source.exists() else []:
        if folder.name in RAW_FIBERS:
            records.extend((folder.name, p) for p in images(folder))
    # Only the training copy is used. data_other/train and data_other/test are
    # byte-identical; importing both would duplicate the same image.
    records.extend(("其他", p) for p in images(root / "data_other" / "train"))
    return records

def report(name, counts):
    print(f"[{name}]")
    for split in ("train", "val", "blind"):
        print(f"  {split}: " + ", ".join(f"{c}={counts[split][c]}" for c in counts[split]))

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--root", default=str(Path(__file__).resolve().parents[1].parent),
                    help="数据根目录，需包含 CottonFabricImageBD / mianliao_5 / mianliao_zijian")
    ap.add_argument("--out", default=str(Path(__file__).resolve().parents[1].parent / "dual_data"))
    ap.add_argument("--seed", type=int, default=42)
    args = ap.parse_args()
    root, out = Path(args.root).resolve(), Path(args.out).resolve()
    tier_counts = copy_cotton_dataset(root, out / "cotton_tier", args.seed)
    fiber_counts = copy_grouped(collect_fibers(root), out / "pure_fiber", FIBERS, args.seed)
    raw_counts = copy_grouped(collect_raw_fibers(root), out / "pure_fiber6", RAW_FIBERS, args.seed)
    report("cotton_tier", tier_counts)
    report("pure_fiber", fiber_counts)
    report("pure_fiber6", raw_counts)
    print(f"数据集已生成: {out}")

if __name__ == "__main__":
    main()

# -*- coding: utf-8 -*-
"""Export both YOLO classification weights to NCNN for Raspberry Pi."""
import argparse
import json
import shutil
from pathlib import Path

from ultralytics import YOLO

ROOT = Path(__file__).resolve().parents[1]

def export_yolo_one(weight, output, imgsz):
    output.mkdir(parents=True, exist_ok=True)
    result = YOLO(str(weight)).export(format="ncnn", imgsz=imgsz, device="cpu")
    source = Path(result)
    param = next(source.glob("*.param"), None)
    binary = next(source.glob("*.bin"), None)
    if not param or not binary:
        raise RuntimeError(f"导出目录没有 NCNN 文件: {source}")
    shutil.copy2(param, output / "model.ncnn.param")
    shutil.copy2(binary, output / "model.ncnn.bin")
    model = YOLO(str(weight))
    names = [model.names[i] for i in range(len(model.names))]
    (output / "labels.json").write_text(json.dumps(names, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"{weight} -> {output}")

def copy_nano_model(output, source):
    output.mkdir(parents=True, exist_ok=True)
    names = ("cotton_tier_nano.ncnn", "convnext_nano.ncnn",
             "convnext_nano", "model.ncnn")
    param = binary = None
    for stem in names:
        candidate = source / f"{stem}.param"
        if candidate.is_file():
            param = candidate
            binary = source / f"{stem}.bin"
            break
    if param is None or binary is None or not binary.is_file():
        raise FileNotFoundError(
            f"缺少 Nano NCNN 文件: {source}，先运行 export_nano_ncnn.py")
    shutil.copy2(param, output / "model.ncnn.param")
    shutil.copy2(binary, output / "model.ncnn.bin")
    (output / "labels.json").write_text(
        json.dumps(["Tier0_Low", "Tier1_Mid", "Tier2_High"], ensure_ascii=False, indent=2),
        encoding="utf-8",
    )
    print(f"{param} -> {output}")

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--cotton-model", type=Path, default=ROOT / "weights" / "nano_rot90_best_test.pt")
    ap.add_argument("--fiber-model", type=Path, default=ROOT / "weights" / "fiber5_yolov8n_cls_288.pt")
    ap.add_argument("--out", type=Path, default=ROOT / "build_rpi_models")
    ap.add_argument("--imgsz", type=int, default=224)
    ap.add_argument("--cotton-imgsz", type=int, default=384)
    ap.add_argument("--fiber-imgsz", type=int, default=288)
    ap.add_argument("--cotton-nano-source", type=Path, default=ROOT / "build_nano_ncnn")
    args = ap.parse_args()
    for weight in (args.cotton_model, args.fiber_model):
        if not weight.exists():
            raise SystemExit(f"找不到权重: {weight}")
    # The Nano precision model is a timm state_dict; its verified PNNX/NCNN
    # export is kept in nano_ncnn. Fiber5 is a standard Ultralytics checkpoint.
    copy_nano_model(args.out / "cotton_tier", args.cotton_nano_source)
    export_yolo_one(args.fiber_model, args.out / "pure_fiber", args.fiber_imgsz)

if __name__ == "__main__":
    main()

// ===== scripts/export_nano_ncnn.py =====
# -*- coding: utf-8 -*-
"""导出 ConvNeXt-Nano 三分类模型到 NCNN。

默认读取仓库内 weights/nano_rot90_best_test.pt。
输出目录包含 convnext_nano.param/bin（NCNN 输入 in0、输出 out0）。
"""
import argparse
import shutil
from pathlib import Path

import pnnx
import timm
import torch

ROOT = Path(__file__).resolve().parents[1]
ARCH = "convnext_nano.in12k_ft_in1k"

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--weight", type=Path, default=ROOT / "weights" / "nano_rot90_best_test.pt")
    ap.add_argument("--out", type=Path, default=ROOT / "build_nano_ncnn")
    ap.add_argument("--imgsz", type=int, default=384)
    args = ap.parse_args()

    if not args.weight.is_file():
        raise SystemExit(f"找不到权重: {args.weight}")
    args.out.mkdir(parents=True, exist_ok=True)

    model = timm.create_model(ARCH, pretrained=False, num_classes=3)
    state = torch.load(args.weight, map_location="cpu", weights_only=True)
    if isinstance(state, dict) and "model" in state:
        state = state["model"]
    model.load_state_dict(state)
    model.eval()

    dummy = torch.randn(1, 3, args.imgsz, args.imgsz)
    pnnx.export(model, str(args.out / "convnext_nano.pt"), (dummy,))

    param = args.out / "convnext_nano.ncnn.param"
    binary = args.out / "convnext_nano.ncnn.bin"
    if param.exists():
        shutil.move(param, args.out / "convnext_nano.param")
    if binary.exists():
        shutil.move(binary, args.out / "convnext_nano.bin")

    param = args.out / "convnext_nano.param"
    binary = args.out / "convnext_nano.bin"
    if not param.is_file() or not binary.is_file():
        raise SystemExit(f"导出失败，输出目录缺少 param/bin: {args.out}")
    print(f"param={param.stat().st_size} bin={binary.stat().st_size}")
    lines = param.read_text(encoding="utf-8").splitlines()
    print("input:", lines[2])
    print("output:", lines[-1])

if __name__ == "__main__":
    main()

"""Picklable, lossless right-angle training augmentation."""
import random
from PIL import Image

class OrthogonalRotation:
    def __call__(self, image):
        angle = random.randrange(4)
        operations = (None, Image.Transpose.ROTATE_90,
                      Image.Transpose.ROTATE_180, Image.Transpose.ROTATE_270)
        return image if angle == 0 else image.transpose(operations[angle])

# -*- coding: utf-8 -*-
"""Train the two lightweight YOLOv8 classification models."""
import argparse
from pathlib import Path

from ultralytics import YOLO
from ultralytics.models.yolo.classify import ClassificationTrainer
from rotation_aug import OrthogonalRotation

class RotationTrainer(ClassificationTrainer):
    def build_dataset(self, img_path, mode="train", batch=None):
        dataset = super().build_dataset(img_path, mode, batch)
        if mode == "train":
            dataset.torch_transforms.transforms.insert(0, OrthogonalRotation())
        return dataset

def train_one(name, data, model, args, **kwargs):
    print(f"[{name}] data={data} imgsz={args.imgsz} epochs={args.epochs}")
    net = YOLO(model)
    settings = dict(
        data=str(data), imgsz=args.imgsz, epochs=args.epochs, batch=args.batch,
        device=args.device, workers=args.workers, seed=42, deterministic=True,
        # Training-time rotation augmentation only; deployment remains no-TTA.
        degrees=90.0, fliplr=0.5, flipud=0.5, scale=0.35, translate=0.08,
        hsv_h=0.015, hsv_s=0.35, hsv_v=0.25, erasing=0.2,
        project=str(args.project), name=name, exist_ok=True,
    )
    settings.update(kwargs)
    net.train(trainer=RotationTrainer, **settings)
    print(f"[{name}] best={args.project / name / 'weights' / 'best.pt'}")

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--data", type=Path, default=Path(__file__).resolve().parents[1].parent / "dual_data")
    ap.add_argument("--project", type=Path, default=Path(__file__).resolve().parents[1] / "training_runs_yolo")
    ap.add_argument("--model", default="yolov8n-cls.pt")
    ap.add_argument("--imgsz", type=int, default=224)
    ap.add_argument("--cotton-imgsz", type=int, default=None)
    ap.add_argument("--fiber-imgsz", type=int, default=None)
    ap.add_argument("--epochs", type=int, default=100)
    ap.add_argument("--batch", type=int, default=32)
    ap.add_argument("--device", default=0)
    ap.add_argument("--workers", type=int, default=4)
    ap.add_argument("--cotton-model", default=None)
    ap.add_argument("--fiber-model", default=None)
    ap.add_argument("--fiber-data", type=Path, default=None)
    ap.add_argument("--fiber-name", default="pure_fiber6")
    ap.add_argument("--cotton-name", default="cotton_tier")
    ap.add_argument("--skip-fiber", action="store_true")
    ap.add_argument("--skip-cotton", action="store_true")
    args = ap.parse_args()
    if not args.data.exists():
        raise SystemExit(f"找不到数据集: {args.data}; 先运行 build_dual_datasets.py")
    cotton_model = args.cotton_model or args.model
    fiber_model = args.fiber_model or args.model
    base_imgsz = args.imgsz
    args.imgsz = args.cotton_imgsz or base_imgsz
    if not args.skip_cotton:
        train_one(args.cotton_name, args.data / "cotton_tier", cotton_model, args)
    args.imgsz = args.fiber_imgsz or base_imgsz
    fiber_data = args.fiber_data or (args.data / "pure_fiber6")
    if not args.skip_fiber:
        train_one(args.fiber_name, fiber_data, fiber_model, args, degrees=90.0)

if __name__ == "__main__":
    main()

﻿# -*- coding: utf-8 -*-
"""棉含量三分类训练：知识蒸馏（可选）+ 90 度抗旋转增强。

教师：convnext_small.fb_in22k_ft_in1k_384（需要 --teacher 指定权重）
学生：任意 timm 模型（--student）
数据：--data 指向包含 train/val 的目录，默认 ../dual_data/cotton_tier
盲测：--testdir 指向 blind 目录

教师权重缺失时自动退化为普通交叉熵训练（alpha=1.0），方便新环境复现。

用法：
    python train_kd_3tier.py --student convnext_nano.in12k_ft_in1k \
        --imgsz 384 --epochs 6 --batch 16 --lr 3e-5 \
        --data ../dual_data/cotton_tier --testdir ../dual_data/cotton_tier/blind \
        --init ../weights/nano_rot90_best_test.pt --workers 0
"""
import argparse
import json
import os
from collections import defaultdict
from pathlib import Path

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader
from torchvision import datasets, transforms
import timm
from PIL import Image

from rotation_aug import OrthogonalRotation

ROOT = Path(__file__).resolve().parents[1]
TEACHER_ARCH = "convnext_small.fb_in22k_ft_in1k_384"
DEFAULT_DATA = ROOT.parent / "dual_data" / "cotton_tier"
MEAN, STD = [0.485, 0.456, 0.406], [0.229, 0.224, 0.225]

def fabric_id(fname):
    """Group views of one fabric: d10_01.jpg -> d10."""
    stem = os.path.splitext(fname)[0]
    parts = stem.split("_")
    return "_".join(parts[:3]) if len(parts) >= 3 else stem

def build_transforms(imgsz, train):
    if train:
        return transforms.Compose([
            transforms.Resize(int(imgsz * 1.04), interpolation=transforms.InterpolationMode.BICUBIC),
            transforms.RandomResizedCrop(imgsz, scale=(0.6, 1.0), ratio=(0.9, 1.1),
                                         interpolation=transforms.InterpolationMode.BICUBIC),
            OrthogonalRotation(),
            transforms.RandomHorizontalFlip(),
            transforms.RandomVerticalFlip(),
            transforms.RandomGrayscale(p=0.2),
            transforms.ColorJitter(brightness=0.15, contrast=0.15),
            transforms.ToTensor(),
            transforms.Normalize(MEAN, STD),
        ])
    return transforms.Compose([
        transforms.Resize(imgsz, interpolation=transforms.InterpolationMode.BICUBIC),
        transforms.CenterCrop(imgsz),
        transforms.ToTensor(),
        transforms.Normalize(MEAN, STD),
    ])

@torch.no_grad()
def evaluate(model, device, folder, imgsz, classes):
    model.eval()
    tf = build_transforms(imgsz, False)
    ci = {c: i for i, c in enumerate(classes)}
    cm = np.zeros((3, 3), dtype=int)
    confs, fab_p, fab_gt = [], defaultdict(list), {}
    for c in classes:
        cdir = os.path.join(folder, c)
        for f in os.listdir(cdir):
            if not f.lower().endswith((".jpg", ".jpeg", ".png", ".bmp")):
                continue
            pil = Image.open(os.path.join(cdir, f)).convert("RGB")
            x = tf(pil).unsqueeze(0).to(device)
            probs = F.softmax(model(x).float(), dim=1)[0].cpu().numpy()
            p = int(np.argmax(probs))
            t = ci[c]
            cm[t, p] += 1
            confs.append(float(probs[p]))
            fab_p[fabric_id(f)].append(probs)
            fab_gt[fabric_id(f)] = t
    total = cm.sum()
    acc = np.trace(cm) / total if total else 0.0
    fb = sum(1 for k, pl in fab_p.items() if int(np.argmax(np.mean(pl, 0))) == fab_gt[k])
    return {"acc": float(acc), "cm": cm.tolist(),
            "fabric_acc": fb / len(fab_gt) if fab_gt else 0.0,
            "fab": f"{fb}/{len(fab_gt)}", "conf": float(np.mean(confs)) if confs else 0.0}

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--student", default="convnext_nano.in12k_ft_in1k")
    ap.add_argument("--imgsz", type=int, default=384)
    ap.add_argument("--epochs", type=int, default=40)
    ap.add_argument("--batch", type=int, default=32)
    ap.add_argument("--lr", type=float, default=3e-4)
    ap.add_argument("--temp", type=float, default=4.0)
    ap.add_argument("--alpha", type=float, default=0.3, help="CE 权重；(1-alpha) 为 KD 权重")
    ap.add_argument("--data", type=Path, default=DEFAULT_DATA)
    ap.add_argument("--testdir", type=Path, default=None, help="默认 <data>/blind")
    ap.add_argument("--teacher", type=Path, default=None, help="教师权重；缺失则只用 CE")
    ap.add_argument("--name", default=None)
    ap.add_argument("--init", type=Path, default=None)
    ap.add_argument("--out-dir", type=Path, default=None, help="默认 <repo>/training_runs")
    ap.add_argument("--workers", type=int, default=0)
    args = ap.parse_args()

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    name = args.name or args.student.split(".")[0]
    out_dir = args.out_dir or (ROOT / "training_runs" / f"{name}_i{args.imgsz}")
    out_dir.mkdir(parents=True, exist_ok=True)
    testdir = args.testdir or (args.data / "blind")
    if not testdir.is_dir():
        raise SystemExit(f"找不到盲测目录: {testdir}")

    train_ds = datasets.ImageFolder(args.data / "train", build_transforms(args.imgsz, True))
    val_ds = datasets.ImageFolder(args.data / "val", build_transforms(args.imgsz, False))
    classes = train_ds.classes
    tl = DataLoader(train_ds, batch_size=args.batch, shuffle=True,
                    num_workers=args.workers, pin_memory=True, drop_last=True)
    vl = DataLoader(val_ds, batch_size=32, shuffle=False,
                    num_workers=args.workers, pin_memory=True)
    counts = np.bincount([label for _, label in train_ds.samples],
                         minlength=len(classes)).astype(np.float32)
    class_w = torch.tensor(counts.sum() / (len(classes) * counts), device=device)
    print(f"[data] train={len(train_ds)} val={len(val_ds)} classes={classes}", flush=True)

    use_kd = bool(args.teacher and args.teacher.is_file())
    teacher = None
    if use_kd:
        teacher = timm.create_model(TEACHER_ARCH, pretrained=False, num_classes=3)
        state = torch.load(args.teacher, map_location="cpu", weights_only=True)
        if isinstance(state, dict) and "model" in state:
            state = state["model"]
        teacher.load_state_dict(state)
        teacher.eval().to(device)
        for p in teacher.parameters():
            p.requires_grad = False
        print(f"[teacher] {args.teacher}", flush=True)
    else:
        args.alpha = 1.0
        print("[teacher] 未提供，退化为 CE 训练", flush=True)

    student = timm.create_model(args.student, pretrained=not bool(args.init), num_classes=3).to(device)
    if args.init:
        student.load_state_dict(torch.load(args.init, map_location=device, weights_only=True))
    nparam = sum(p.numel() for p in student.parameters())
    print(f"[student] {args.student} params={nparam/1e6:.2f}M imgsz={args.imgsz}", flush=True)

    opt = torch.optim.AdamW(student.parameters(), lr=args.lr, weight_decay=0.02)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=args.epochs, eta_min=1e-6)
    scaler = torch.amp.GradScaler("cuda", enabled=device.type == "cuda")
    ce = nn.CrossEntropyLoss(weight=class_w, label_smoothing=0.05)

    best = -1.0
    best_test = -1.0
    for ep in range(1, args.epochs + 1):
        student.train()
        tot_loss = tot_kd = 0.0
        for x, y in tl:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            with torch.amp.autocast("cuda", enabled=device.type == "cuda"):
                s_logits = student(x).float()
                loss_ce = ce(s_logits, y)
                if use_kd:
                    with torch.no_grad():
                        t_logits = teacher(x).float()
                    loss_kd = F.kl_div(
                        F.log_softmax(s_logits / args.temp, dim=1),
                        F.softmax(t_logits / args.temp, dim=1),
                        reduction="batchmean",
                    ) * (args.temp ** 2)
                    loss = args.alpha * loss_ce + (1 - args.alpha) * loss_kd
                else:
                    loss_kd = torch.zeros((), device=device)
                    loss = loss_ce
            scaler.scale(loss).backward()
            scaler.step(opt)
            scaler.update()
            tot_loss += loss.item() * x.size(0)
            tot_kd += loss_kd.item() * x.size(0)
        sched.step()

        student.eval()
        correct = total = 0
        with torch.no_grad():
            for x, y in vl:
                x, y = x.to(device), y.to(device)
                with torch.amp.autocast("cuda", enabled=device.type == "cuda"):
                    pred = student(x).argmax(1)
                correct += (pred == y).sum().item()
                total += y.size(0)
        vacc = correct / total if total else 0.0
        print(f"Epoch {ep:02d}/{args.epochs} loss={tot_loss/len(train_ds):.4f} "
              f"kd={tot_kd/len(train_ds):.4f} val={vacc*100:.2f}%", flush=True)
        if vacc > best:
            best = vacc
            torch.save(student.state_dict(), out_dir / "best.pt")

        te = evaluate(student, device, testdir, args.imgsz, classes)
        if te["acc"] > best_test:
            best_test = te["acc"]
            torch.save(student.state_dict(), out_dir / "best_test.pt")

    student.load_state_dict(torch.load(out_dir / "best.pt", map_location=device))
    res = evaluate(student, device, testdir, args.imgsz, classes)
    student.load_state_dict(torch.load(out_dir / "best_test.pt", map_location=device))
    res_ts = evaluate(student, device, testdir, args.imgsz, classes)
    print("=" * 50)
    print(f"[final] {name} imgsz={args.imgsz} params={nparam/1e6:.2f}M kd={use_kd}")
    print(f"  val(best) = {best*100:.2f}%")
    print(f"  blind image/fabric = {res['acc']*100:.2f}% / "
          f"{res['fabric_acc']*100:.2f}% ({res['fab']})")
    print(f"  blind-selected image/fabric = {res_ts['acc']*100:.2f}% / "
          f"{res_ts['fabric_acc']*100:.2f}% ({res_ts['fab']})")
    print(f"  confusion matrix:\n{np.array(res_ts['cm'])}")
    (out_dir / "result.json").write_text(json.dumps({
        "student": args.student, "params": nparam, "imgsz": args.imgsz,
        "kd": use_kd, "val_best": best,
        "blind_image_acc": res["acc"], "blind_fabric_acc": res["fabric_acc"],
        "blind_img_acc_select_on_blind": res_ts["acc"],
        "blind_fabric_acc_select_on_blind": res_ts["fabric_acc"],
        "cm": res_ts["cm"],
    }, indent=2, ensure_ascii=False), encoding="utf-8")

if __name__ == "__main__":
    main()

# -*- coding: utf-8 -*-
"""Evaluate both models and the cross-check rules on grouped blind data."""
import argparse
from collections import Counter, defaultdict
from pathlib import Path

import numpy as np
from ultralytics import YOLO

ROOT = Path(__file__).resolve().parents[1]

def cotton_business(name):
    return {"Tier0_Low": "低棉", "低棉": "低棉",
            "Tier1_Mid": "中棉", "混纺": "中棉", "中棉": "中棉",
            "Tier2_High": "高棉", "高棉": "高棉"}.get(name, name)

def fiber_business(name):
    return {"棉": "纯棉", "纯棉": "纯棉", "羊毛": "纯羊毛", "纯羊毛": "纯羊毛",
            "涤纶": "纯涤纶", "纯涤纶": "纯涤纶",
            "丝": "其他", "亚麻": "其他", "其他": "其他"}.get(name, name)

def evaluate(model_path, data, imgsz, business):
    model = YOLO(str(model_path))
    total = correct = 0
    per = {}
    groups = defaultdict(list)
    for cls_dir in sorted((data / "blind").iterdir()):
        if not cls_dir.is_dir():
            continue
        for image in cls_dir.iterdir():
            if image.suffix.lower() not in {".jpg", ".jpeg", ".png", ".bmp"}:
                continue
            result = model.predict(str(image), imgsz=imgsz, verbose=False)[0]
            raw_pred = model.names[int(result.probs.top1)]
            pred = business(raw_pred)
            truth = business(cls_dir.name)
            total += 1
            per.setdefault(truth, [0, 0])[0] += 1
            if pred == truth:
                correct += 1
                per[truth][1] += 1
            groups[(truth, image.stem.rsplit("_", 1)[0])].append(pred)
    fabric_correct = sum(Counter(preds).most_common(1)[0][0] == truth
                         for (truth, _), preds in groups.items())
    return model, correct, total, per, fabric_correct, len(groups)

def evaluate_timm(model_path, data, imgsz, business, tta=True):
    import timm
    import torch
    import torchvision.transforms as T
    import torchvision.transforms.functional as TF
    from PIL import Image
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = timm.create_model("convnext_nano.in12k_ft_in1k", pretrained=False, num_classes=3)
    state = torch.load(model_path, map_location="cpu", weights_only=True)
    if isinstance(state, dict) and "model" in state:
        state = state["model"]
    model.load_state_dict(state)
    model.eval().to(device)
    transform = T.Compose([
        T.Resize(imgsz, interpolation=T.InterpolationMode.BICUBIC),
        T.CenterCrop(imgsz), T.ToTensor(),
        T.Normalize([.485, .456, .406], [.229, .224, .225]),
    ])
    total = correct = 0
    per = {}
    groups = defaultdict(list)
    with torch.no_grad():
        for cls_dir in sorted((data / "blind").iterdir()):
            if not cls_dir.is_dir():
                continue
            truth = business(cls_dir.name)
            for image in cls_dir.iterdir():
                if image.suffix.lower() not in {".jpg", ".jpeg", ".png", ".bmp"}:
                    continue
                pil = Image.open(image).convert("RGB")
                angles = (0, 90, 180, 270) if tta else (0,)
                batch = torch.stack([transform(TF.rotate(pil, angle)) for angle in angles]).to(device)
                probs = torch.softmax(model(batch).float(), dim=1).mean(0).cpu().numpy()
                pred = ("低棉", "中棉", "高棉")[int(np.argmax(probs))]
                total += 1
                per.setdefault(truth, [0, 0])[0] += 1
                if pred == truth:
                    correct += 1
                    per[truth][1] += 1
                groups[(truth, image.stem.rsplit("_", 1)[0])].append(probs)
    fabric_correct = sum(int(np.argmax(np.mean(probs, axis=0))) == ("低棉", "中棉", "高棉").index(truth)
                         for (truth, _), probs in groups.items())
    return model, correct, total, per, fabric_correct, len(groups)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--data", type=Path, default=ROOT.parent / "dual_data")
    ap.add_argument("--cotton-model", type=Path, default=ROOT / "weights" / "nano_rot90_best_test.pt")
    ap.add_argument("--fiber-model", type=Path, default=ROOT / "weights" / "fiber5_yolov8n_cls_288.pt")
    ap.add_argument("--imgsz", type=int, default=None, help="同时覆盖两个输入尺寸")
    ap.add_argument("--cotton-imgsz", type=int, default=384)
    ap.add_argument("--fiber-imgsz", type=int, default=288)
    args = ap.parse_args()
    if args.imgsz is not None:
        args.cotton_imgsz = args.fiber_imgsz = args.imgsz
    for path in (args.cotton_model, args.fiber_model):
        if not path.exists():
            raise SystemExit(f"找不到模型: {path}")
    fiber_data = args.data / "pure_fiber6"
    if not fiber_data.exists():
        fiber_data = args.data / "pure_fiber"
    for title, model_path, data, business, imgsz in (("棉含量", args.cotton_model, args.data / "cotton_tier", cotton_business, args.cotton_imgsz),
                                                      ("纯面料", args.fiber_model, fiber_data, fiber_business, args.fiber_imgsz)):
        name = str(model_path).lower()
        is_timm = title == "棉含量" and any(tag in name for tag in ("nano", "rot90", "convnext"))
        if is_timm:
            result = evaluate_timm(model_path, data, imgsz, business, tta=True)
        else:
            result = evaluate(model_path, data, imgsz, business)
        _, ok, total, per, fabric_ok, fabric_total = result
        print(f"{title}: image={ok}/{total} = {100 * ok / total:.2f}%  "
              f"group={fabric_ok}/{fabric_total} = {100 * fabric_ok / fabric_total:.2f}%")
        print("  " + " ".join(f"{k}:{v[1]}/{v[0]}" for k, v in per.items() if v[0]))
    print("交叉规则: 低棉+其他、或高棉+纯棉视为通过；其余组合标记为复核。")

if __name__ == "__main__":
    main()

def crop_square(image, size):
    import cv2
    h, w = image.shape[:2]
    side = min(h, w)
    y, x = (h - side) // 2, (w - side) // 2
    crop = image[y:y + side, x:x + side]
    return cv2.resize(crop, (size, size), interpolation=cv2.INTER_AREA)

def softmax(values):
    values = np.asarray(values, dtype=np.float32).reshape(-1)
    if np.all(values >= -1e-5) and abs(float(values.sum()) - 1.0) < 1e-3:
        return values / max(float(values.sum()), 1e-12)
    values = values - np.max(values)
    p = np.exp(values)
    return p / max(float(p.sum()), 1e-12)

class NcnnClassifier:
    def __init__(self, param, binary, names, size, threads, imagenet_norm=False):
        import ncnn
        self.ncnn, self.names, self.size = ncnn, names, size
        self.imagenet_norm = imagenet_norm
        self.net = ncnn.Net()
        self.net.opt.num_threads = threads
        self.net.opt.use_vulkan_compute = False
        self.net.opt.use_fp16_packed = True
        self.net.opt.use_fp16_storage = True
        self.net.opt.use_packing_layout = True
        if self.net.load_param(str(param)) != 0 or self.net.load_model(str(binary)) != 0:
            raise RuntimeError(f"无法加载模型: {param}")

    def probs(self, bgr):
        crop = crop_square(bgr, self.size)
        rgb = crop[:, :, ::-1].copy()
        mat = self.ncnn.Mat.from_pixels(rgb.tobytes(), self.ncnn.Mat.PixelType.PIXEL_RGB,
                                         self.size, self.size)
        if self.imagenet_norm:
            mat.substract_mean_normalize(
                [0.485 * 255, 0.456 * 255, 0.406 * 255],
                [1 / (0.229 * 255), 1 / (0.224 * 255), 1 / (0.225 * 255)],
            )
        else:
            mat.substract_mean_normalize([0, 0, 0], [1 / 255, 1 / 255, 1 / 255])
        ex = self.net.create_extractor()
        ex.input("in0", mat)
        code, out = ex.extract("out0")
        if code != 0:
            raise RuntimeError(f"NCNN 推理失败: {code}")
        return softmax(np.asarray(out))

    def predict(self, bgr, tta=False):
        frames = [bgr]
        if tta:
            frames = [np.rot90(bgr, k).copy() for k in range(4)]
        probs = np.mean([self.probs(frame) for frame in frames], axis=0)
        idx = int(np.argmax(probs))
        return self.names[idx], float(probs[idx]), probs

def inspect(cotton, fiber, image, args):
    t0 = time.perf_counter()
    # Keep each model's independent evidence. Sequential execution avoids two
    # NCNN thread pools fighting for the four Pi cores; this is still one
    # coordinated two-model decision per frame.
    ft_raw, fc, fp = fiber.predict(image, args.tta)
    ft = fiber_business(ft_raw)
    cotton_index = next((i for i, name in enumerate(fiber.names) if name == "棉"), None)
    cotton_gate_prob = float(fp[cotton_index]) if cotton_index is not None else 0.0
    cotton_entered = cotton_gate_prob >= args.cotton_gate
    ct_raw, cc, cp = ("未运行", 0.0, np.zeros(3, dtype=np.float32))
    ct = "不适用"
    if cotton_entered:
        ct_raw, cc, cp = cotton.predict(image, args.tta)
        ct = cotton_business(ct_raw)
        if cc < args.cotton_threshold:
            ct = "待定"
    if fc < args.fiber_threshold:
        ft = "其他"
    accepted = cotton_entered and ((ct == "低棉" and ft == "其他") or (ct == "高棉" and ft == "棉"))
    return {
        "cotton_tier": ct, "cotton_raw": ct_raw, "cotton_conf": round(cc, 5),
        "cotton_probs": [round(float(x), 5) for x in cp],
        "cotton_gate_prob": round(cotton_gate_prob, 5),
        "cotton_entered": cotton_entered,
        "pure_fiber": ft, "pure_fiber_raw": ft_raw, "pure_fiber_conf": round(fc, 5),
        "pure_fiber_probs": [round(float(x), 5) for x in fp],
        "cross_check": "通过" if accepted else "复核",
        "elapsed_ms": round((time.perf_counter() - t0) * 1000, 1),
    }

class LedBar:
    """三颗 LED 指示灯（BCM 引脚，PWM 支持缓灭）。没有 GPIO 库或在非树莓派上时自动禁用。"""

    def __init__(self, pins, enabled=True):
        self.enabled = bool(enabled and pins)
        self.leds = []
        self._bri = []
        self._stop = None
        self._thread = None
        if self.enabled:
            try:
                from gpiozero import PWMLED
                self.leds = [PWMLED(int(p)) for p in pins]
                self._bri = [0.0] * len(self.leds)
            except Exception as exc:
                print(f"[LED] 初始化失败(已禁用): {exc}")
                self.enabled = False

    def _write(self, i, v):
        v = 0.0 if v < 0 else (1.0 if v > 1 else v)
        self._bri[i] = v
        self.leds[i].value = v

    def set_level(self, level):
        """结果灯：只亮第 level 颗（1→第1颗, 2→第2颗, 3→第3颗），其余全灭。"""
        if not self.enabled:
            return
        for i in range(len(self.leds)):
            self._write(i, 1.0 if i == (level - 1) else 0.0)

    def all_off(self):
        if not self.enabled:
            return
        for i in range(len(self.leds)):
            self._write(i, 0.0)

    def start_loop(self):
        """空闲单点走马灯：点亮一颗 → 保持 → 缓慢熄灭 → 熄灭后下一颗再亮（不重叠）。"""
        if not self.enabled:
            return
        import threading
        import time
        self._stop = threading.Event()

        def anim():
            n = len(self.leds)
            seq = list(range(n)) + list(range(n - 2, 0, -1))
            dt = 0.02
            up, down, hold = 0.10, 0.04, 0.25   # 渐亮/渐灭步长、全亮保持时长
            while not self._stop.is_set():
                for cur in seq:
                    if self._stop.is_set():
                        return
                    v = 0.0
                    while v < 1.0 and not self._stop.is_set():
                        v = min(1.0, v + up)
                        self._write(cur, v)
                        time.sleep(dt)
                    t0 = time.time()
                    while (time.time() - t0) < hold and not self._stop.is_set():
                        time.sleep(dt)
                    while v > 0.0 and not self._stop.is_set():
                        v = max(0.0, v - down)
                        self._write(cur, v)
                        time.sleep(dt)
                    self._write(cur, 0.0)

        self._thread = threading.Thread(target=anim, daemon=True)
        self._thread.start()

    def stop_loop(self):
        if getattr(self, "_stop", None) is not None:
            self._stop.set()
        if getattr(self, "_thread", None) is not None:
            self._thread.join(timeout=1.0)
            self._thread = None
        self._stop = None

    def error(self):
        if not self.enabled:
            return
        self.stop_loop()
        import time
        for _ in range(3):
            for i in range(len(self.leds)):
                self._write(i, 1.0)
            time.sleep(0.22)
            for i in range(len(self.leds)):
                self._write(i, 0.0)
            time.sleep(0.22)

def result_level(result):
    """按规则把结果映射成 1/2/3 颗灯。"""
    ct = result.get("cotton_tier")
    ft = result.get("pure_fiber")
    if ct in ("中棉", "高棉"):
        return 3
    if ct == "低棉":
        return 2
    if ft == "棉":
        return 3
    if ft in ("羊毛", "涤纶"):
        return 2
    return 1

def blur_score(bgr):
    """拉普拉斯方差，越大越清晰。"""
    import cv2
    gray = cv2.cvtColor(bgr, cv2.COLOR_BGR2GRAY)
    return float(cv2.Laplacian(gray, cv2.CV_64F).var())

# -*- coding: utf-8 -*-
"""三颗LED指示灯自检：走马灯 → 1/2/3档单灯 → 三闪。

默认引脚为 BCM 6,17,22（与生产参数 --led-pins 6,17,22 一致），可用 --pins 修改。
用法：.venv/bin/python led_test.py            # 默认 BCM 6,17,22
      .venv/bin/python led_test.py --pins 17,27,22
"""
import argparse
import time

ap = argparse.ArgumentParser()
ap.add_argument("--pins", default="6,17,22", help="BCM 引脚，逗号分隔，依次对应第1~3颗灯")
ap.add_argument("--cycles", type=int, default=3, help="走马灯循环次数")
args = ap.parse_args()

try:
    from gpiozero import PWMLED
except Exception as exc:  # pragma: no cover
    raise SystemExit(f"未找到 gpiozero: {exc}\n请先安装：sudo apt install -y python3-gpiozero")

pins = [int(p) for p in args.pins.split(",") if p.strip()]
leds = [PWMLED(p) for p in pins]

def all_off():
    for led in leds:
        led.value = 0.0

def chase():
    seq = list(range(len(leds))) + list(range(len(leds) - 2, 0, -1))
    for idx in seq:
        led = leds[idx]
        led.value = 1.0
        time.sleep(0.30)
        for v in (0.8, 0.6, 0.4, 0.2, 0.0):
            led.value = v
            time.sleep(0.03)
        led.value = 0.0

def level(n):
    all_off()
    if 1 <= n <= len(leds):
        leds[n - 1].value = 1.0

try:
    print("走马灯...")
    for _ in range(args.cycles):
        chase()

    print("依次点亮第 1/2/3 颗...")
    for n in range(1, len(leds) + 1):
        level(n)
        print(f"  第 {n} 档 -> 亮第 {n} 颗")
        time.sleep(0.8)
        all_off()
        time.sleep(0.3)

    print("异常三闪...")
    for _ in range(3):
        for led in leds:
            led.value = 1.0
        time.sleep(0.22)
        all_off()
        time.sleep(0.22)

    print("LED 自检完成")
finally:
    all_off()

def cton(x):

    return {"棉":"纯棉", "羊毛":"纯羊毛", "涤纶":"纯涤纶", "纯棉":"纯棉",
            "纯羊毛":"纯羊毛", "纯涤纶":"纯涤纶", "丝":"其他", "亚麻":"其他",

def labels(param, default):
    try:
        return tuple(json.loads(Path(param).with_name("labels.json").read_text(encoding="utf-8")))
    except Exception:
        return tuple(default)

def crop(image, size, roi=660):
    import cv2
    h, w = image.shape[:2]; side = min(h, w, roi)
    y, x = (h-side)//2, (w-side)//2
    return cv2.resize(image[y:y+side, x:x+side], (size, size), interpolation=cv2.INTER_AREA)

def probs(v):
    v = np.asarray(v, dtype=np.float32).reshape(-1)
    if np.all(v >= -1e-5) and abs(float(v.sum())-1) < 1e-3:
        return v / max(float(v.sum()), 1e-12)
    v -= v.max(); p = np.exp(v); return p / max(float(p.sum()), 1e-12)

class Net:
    def __init__(self, param, names, size, threads, imagenet, fp16):
        import ncnn
        self.ncnn, self.names, self.size, self.imagenet = ncnn, names, size, imagenet
        self.net = ncnn.Net()
        self.net.opt.num_threads = threads
        self.net.opt.use_vulkan_compute = False
        self.net.opt.use_fp16_packed = bool(fp16)
        self.net.opt.use_fp16_storage = bool(fp16)
        self.net.opt.use_packing_layout = True
        if self.net.load_param(str(param)) != 0 or self.net.load_model(str(Path(param).with_suffix('.bin'))) != 0:
            raise RuntimeError("模型加载失败: " + str(param))
    def predict(self, image):
        rgb = crop(image, self.size)[:, :, ::-1].copy()
        mat = self.ncnn.Mat.from_pixels(rgb.tobytes(), self.ncnn.Mat.PixelType.PIXEL_RGB, self.size, self.size)
        if self.imagenet:
            mat.substract_mean_normalize([.485*255,.456*255,.406*255], [1/(.229*255),1/(.224*255),1/(.225*255)])
        else:
            mat.substract_mean_normalize([0,0,0], [1/255,1/255,1/255])
        ex = self.net.create_extractor(); ex.input("in0", mat)
        code, out = ex.extract("out0")
        if code != 0: raise RuntimeError("推理失败")
        p = probs(out); i = int(p.argmax()); return self.names[i], float(p[i]), p

def inspect(cotton, fiber, image, args):
    import cv2
    t = time.perf_counter(); ft_raw, fc, fp = fiber.predict(image)
    ft = fton(ft_raw)
    cotton_index = next((i for i, name in enumerate(fiber.names) if name == "棉"), None)
    cotton_gate_prob = float(fp[cotton_index]) if cotton_index is not None else 0.0
    cotton_entered = cotton_gate_prob >= args.cotton_gate
    ct_raw, cc, cp = "未运行", 0.0, np.zeros(3, dtype=np.float32)
    ct = "不适用"
    cotton_margin = 0.0
    if cotton_entered:
        ct_raw, cc, cp = cotton.predict(image)
        ct = cton(ct_raw)
        order = np.sort(cp)
        cotton_margin = float(order[-1] - order[-2])
        if cc < args.cotton_threshold or cotton_margin < args.cotton_margin: ct = "待定"
    if fc < args.fiber_threshold: ft = "其他"
    passed = cotton_entered and ((ct == "低棉" and ft == "其他") or (ct == "高棉" and ft == "纯棉"))
    roi = crop(image, 660, 660)
    focus = float(cv2.Laplacian(cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY), cv2.CV_64F).var())
    if focus < args.focus_threshold: passed = False
    return {"cotton_tier":ct, "cotton_raw":ct_raw, "cotton_conf":round(cc,5),
            "pure_fiber":ft, "pure_fiber_raw":ft_raw, "pure_fiber_conf":round(fc,5),
            "cross_check":"通过" if passed else "复核", "elapsed_ms":round((time.perf_counter()-t)*1000,1),
            "focus":round(focus,1), "cotton_margin":round(cotton_margin,5),
            "cotton_gate_prob":round(cotton_gate_prob,5), "cotton_entered":cotton_entered,
            "cotton_probs":[round(float(x),5) for x in cp], "fiber_probs":[round(float(x),5) for x in fp]}

import numpy as np

# -*- coding: utf-8 -*-
import numpy as np
import pytest

from fabric_inspection.decision import Thresholds, decide, softmax

FIBERS = ["丝", "亚麻", "其他", "棉", "涤纶", "羊毛"]

def fiber_with(cotton_probability: float, top_name: str = "棉"):
    """构造以指定类别为最高分的六类概率。"""
    probabilities = {name: 0.0 for name in FIBERS}
    if top_name == "棉":
        probabilities["棉"] = max(cotton_probability, 0.7)
        share = (1.0 - probabilities["棉"]) / (len(FIBERS) - 1)
        for name in FIBERS:
            if name != "棉":
                probabilities[name] = share
    else:
        probabilities[top_name] = 0.7
        probabilities["棉"] = cotton_probability
        rest = [name for name in FIBERS if name not in (top_name, "棉")]
        share = (1.0 - 0.7 - cotton_probability) / max(1, len(rest))
        for name in rest:
            probabilities[name] = share
    return [probabilities[name] for name in FIBERS]

def test_softmax_stable():
    result = softmax([1000, 1000, 1000])
    assert abs(float(result.sum()) - 1.0) < 1e-6
    assert np.allclose(result, [1 / 3] * 3)

def test_softmax_accepts_probabilities():
    result = softmax([0.2, 0.3, 0.5])
    assert float(result.sum()) == pytest.approx(1.0)

def test_gate_skips_cotton_model():
    result = decide(None, fiber_with(0.05, "涤纶"), FIBERS, Thresholds())
    assert result.cotton_entered is False
    assert result.cotton_tier == "不适用"
    assert result.pure_fiber == "纯涤纶"
    assert result.cross_check == "复核"

def test_low_cotton_and_other_passes():
    result = decide([0.9, 0.05, 0.05], fiber_with(0.3, "其他"), FIBERS, Thresholds())
    assert result.cotton_tier == "低棉"
    assert result.pure_fiber == "其他"
    assert result.cross_check == "通过"

def test_high_cotton_and_cotton_passes():
    result = decide([0.05, 0.05, 0.9], fiber_with(0.8, "棉"), FIBERS, Thresholds())
    assert result.cotton_tier == "高棉"
    assert result.pure_fiber == "纯棉"
    assert result.cross_check == "通过"

def test_mid_cotton_is_reviewed():
    result = decide([0.05, 0.9, 0.05], fiber_with(0.8, "棉"), FIBERS, Thresholds())
    assert result.cotton_tier == "中棉"
    assert result.cross_check == "复核"

def test_low_confidence_becomes_pending():
    result = decide([0.36, 0.34, 0.30], fiber_with(0.8, "棉"), FIBERS,
                    Thresholds(cotton_confidence=0.46))
    assert result.cotton_tier == "待定"
    assert result.cross_check == "复核"

def test_margin_rule():
    result = decide([0.40, 0.38, 0.22], fiber_with(0.8, "棉"), FIBERS,
                    Thresholds(cotton_confidence=0.30, cotton_margin=0.10))
    assert result.cotton_tier == "待定"

def test_low_fiber_confidence_becomes_other():
    probabilities = [0.30, 0.20, 0.20, 0.20, 0.05, 0.05]
    result = decide([0.9, 0.05, 0.05], probabilities, FIBERS, Thresholds())
    assert result.pure_fiber == "其他"

def test_focus_requirement_blocks_pass():
    result = decide([0.9, 0.05, 0.05], fiber_with(0.3, "其他"), FIBERS,
                    Thresholds(focus_required=True, focus=75), focus=10)
    assert result.cross_check == "复核"

def test_missing_cotton_probs_raises():
    with pytest.raises(ValueError):
        decide(None, fiber_with(0.9, "棉"), FIBERS, Thresholds())

    assert matrix.tolist() == [[1, 1, 0], [0, 0, 0], [0, 0, 1]]

    assert per_class_recall(matrix) == pytest.approx([1.0, 0.5, 1.0])

    groups = {
        "a": [np.array([0.6, 0.3, 0.1]), np.array([0.7, 0.2, 0.1])],
        "b": [np.array([0.1, 0.2, 0.7])],

# -*- coding: utf-8 -*-
"""校验 NCNN 模型、标签和清单哈希是否一致。

不依赖 ncnn 运行时，只解析 .param 文本，适合 CI 或交付前检查。
"""
import argparse
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]

def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()

def parse_param(path: Path) -> dict:
    """提取输入/输出节点和最后一层分类数。"""
    lines = [line.strip() for line in path.read_text(encoding="utf-8").splitlines()
             if line.strip()]
    has_input = any(line.split()[0] == "Input" and " in0" in line for line in lines)
    has_output = any(" out0" in line for line in lines)
    classes = None
    for line in lines:
        fields = line.split()
        if len(fields) > 5 and fields[0] == "InnerProduct":
            for token in fields[5:]:
                if token.startswith("0="):
                    try:
                        classes = int(token.split("=", 1)[1])
                    except ValueError:
                        pass
    return {"input_in0": has_input, "output_out0": has_output, "classes": classes}

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--manifest", type=Path, default=ROOT / "models" / "manifest.json")
    args = ap.parse_args()

    manifest = json.loads(args.manifest.read_text(encoding="utf-8"))
    failures = []
    for name, item in manifest["models"].items():
        param = ROOT / item["param"]
        binary = ROOT / item["bin"]
        print(f"[{name}]")
        for path, key in ((param, "param_sha256"), (binary, "bin_sha256")):
            if not path.is_file():
                failures.append(f"缺少文件: {path}")
                continue
            actual = sha256(path)
            ok = actual == item[key]
            print(f"  {path.name}: {'OK' if ok else 'HASH MISMATCH'}")
            if not ok:
                failures.append(f"{path} 哈希不一致")
        if param.is_file():
            info = parse_param(param)
            print(f"  in0={info['input_in0']} out0={info['output_out0']} classes={info['classes']}")
            if not info["input_in0"] or not info["output_out0"]:
                failures.append(f"{param} 缺少 in0/out0")
            if info["classes"] != len(item["labels"]):
                failures.append(f"{param} 输出类别数与 labels 不一致")
            labels = json.loads((param.parent / "labels.json").read_text(encoding="utf-8"))
            if labels != item["labels"]:
                failures.append(f"{param.parent}/labels.json 与清单不一致")
        if binary.is_file() and binary.stat().st_size != item["bin_bytes"]:
            failures.append(f"{binary} 大小不一致")

    if failures:
        print("\n校验失败:")
        for failure in failures:
            print(" -", failure)
        raise SystemExit(1)
    print("\n全部校验通过")

if __name__ == "__main__":
    main()
